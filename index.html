SIMPLE ARBITRAGE OPTIMISER # ------------------------------------------------------------ def optimise_forecast( prices, power_mw=100, energy_mwh=200, efficiency=0.90, initial_soc=None ): """ Optimises energy arbitrage using ONLY the supplied forecast prices. Returns a 24-hour dispatch schedule. """ prices = np.asarray(prices, dtype=float) n = len(prices) if initial_soc is None: initial_soc = energy_mwh * 0.5 soc = initial_soc charge = np.zeros(n) discharge = np.zeros(n) soc_path = np.zeros(n) revenue = np.zeros(n) # -------------------------------------------------------- # Rank hours by forecast price # -------------------------------------------------------- cheap_hours = np.argsort(prices) expensive_hours = np.argsort(prices)[::-1] used_charge = set() used_discharge = set() # Maximum one full equivalent cycle per day remaining_energy = energy_mwh # -------------------------------------------------------- # Greedy forecast-based dispatch # -------------------------------------------------------- for buy_hour in cheap_hours: if remaining_energy <= 0: break for sell_hour in expensive_hours: if sell_hour <= buy_hour: continue if sell_hour in used_discharge: continue if buy_hour in used_charge: continue spread = prices[sell_hour] - prices[buy_hour] if spread <= 0: continue charge_energy = min( power_mw, energy_mwh - soc, remaining_energy ) if charge_energy <= 0: continue discharge_energy = ( charge_energy * efficiency ) discharge_energy = min( discharge_energy, power_mw, energy_mwh ) charge[buy_hour] = charge_energy discharge[sell_hour] = discharge_energy soc += charge_energy * efficiency soc -= discharge_energy used_charge.add(buy_hour) used_discharge.add(sell_hour) remaining_energy -= charge_energy break # -------------------------------------------------------- # Calculate SOC and revenue properly # -------------------------------------------------------- soc = initial_soc for i in range(n): # Charging costs money if charge[i] > 0: energy_into_battery = ( charge[i] * efficiency ) soc += energy_into_battery revenue[i] -= ( charge[i] * prices[i] ) # Discharging generates revenue if discharge[i] > 0: soc -= discharge[i] revenue[i] += ( discharge[i] * prices[i] ) soc = max( 0, min(energy_mwh, soc) ) soc_path[i] = soc result = pd.DataFrame({ "price_forecast": prices, "charge_mw": charge, "discharge_mw": discharge, "soc_mwh": soc_path, "forecast_revenue": revenue }) return result # ------------------------------------------------------------ # 3. WALK-FORWARD BACKTEST # ------------------------------------------------------------ def walk_forward_backtest( data, power_mw=100, energy_mwh=200, efficiency=0.90, minimum_history_days=14 ): """ Walk-forward simulation. At each day: historical data ↓ forecast tomorrow ↓ optimise forecast ↓ execute against ACTUAL tomorrow prices Future prices are never exposed to the forecast. """ data = data.copy() data["timestamp"] = pd.to_datetime( data["timestamp"] ) data = data.sort_values( "timestamp" ).reset_index(drop=True) data = data.dropna( subset=["price"] ) dates = ( data["timestamp"] .dt.normalize() .drop_duplicates() .sort_values() ) all_results = [] start_index = minimum_history_days for date in dates[start_index:]: history = data[ data["timestamp"] < date ] actual_day = data[ data["timestamp"].dt.normalize() == date ] if len(actual_day) != 24: continue # ---------------------------------------------------- # Execute that dispatch against ACTUAL prices # ---------------------------------------------------- actual_prices = actual_day[ "price" ].values forecast_dispatch["actual_price"] = actual_prices actual_revenue = ( -forecast_dispatch["charge_mw"] * actual_prices + forecast_dispatch["discharge_mw"] * actual_prices ) forecast_dispatch[ "actual_revenue" ] = actual_revenue forecast_profit = actual_revenue.sum() # ---------------------------------------------------- # Perfect foresight # ---------------------------------------------------- perfect_profit = perfect_foresight_day( actual_prices, power_mw=power_mw, energy_mwh=energy_mwh, efficiency=efficiency ) all_results.append({ "date": date, "forecast_profit": forecast_profit, "perfect_profit": perfect_profit }) results = pd.DataFrame(all_results) if len(results) == 0: raise ValueError( "Not enough historical data for walk-forward test." ) #
